return {time:`${h<12?'오전':'오후'} ${h%12||12}:${num(d.getUTCMinutes())}`,day:`${d.getUTCMonth()+1}월 ${d.getUTCDate()}일`,when,same:day===date};
}
function formatClock(minutes,date){const c=clockParts(minutes,date);return `${c.day} ${c.time} (${c.when})`;}
function duration(m){const h=Math.floor(m/60),r=m%60;return h?`${h}시간${r?' '+r+'분':''}`:`${r}분`;}
function longDuration(m){const d=Math.floor(m/1440),r=m%1440;return d?`${d}일${r?' '+duration(r):''}`:duration(r);}
function nowMinutes(){const n=new Date();return parseAppointment(dateString(n),`${num(n.getHours())}:${num(n.getMinutes())}`);}
function calculate(record){const arrival=parseAppointment(record.date,record.time),buffer=record.buffer||0,prep=record.tasks.reduce((sum,t)=>sum+t.min,0),departure=arrival-record.travel-buffer;return {arrival,start:departure-prep,departure,prep,buffer,total:prep+record.travel+buffer};}
// 지금 시각과 비교해 늦었는지 알려 주기 (지난 날짜를 잘못 고른 경우도 잡아냄)
function timing(p,r){
const now=nowMinutes();
if(p.arrival<=now)return ['warn','약속 시각이 이미 지났어요. 날짜와 시각을 다시 확인해 주세요.'];
if(p.start>now)return ['ok',`지금부터 ${longDuration(p.start-now)} 뒤에 준비를 시작하면 돼요.`];
if(p.start===now)return ['ok','지금 바로 준비를 시작하면 딱 맞아요.'];
if(p.departure>now)return ['warn',`준비 시작 시각이 ${longDuration(now-p.start)} 지났어요. 출발까지 ${longDuration(p.departure-now)} 남았으니 준비 항목을 줄여 보세요.`];
const late=now+r.travel-p.arrival;
return ['warn',late>0?`출발 시각이 지났어요. 지금 바로 출발해도 약 ${longDuration(late)} 늦을 수 있어요.`:'출발 시각이 지났어요. 지금 바로 출발하면 약속 시각에는 맞출 수 있어요.'];
}

function planText(r,p){
const lines=[`${clockParts(p.arrival,r.date).day} 약속 준비 계획`,`준비 시작 ${formatClock(p.start,r.date)}`];
let at=p.start;for(const t of r.tasks){lines.push(`· ${clockParts(at,r.date).time} ${t.name} (${duration(t.min)})`);at+=t.min;}
lines.push(`출발 ${formatClock(p.departure,r.date)}`,`약속 ${formatClock(p.arrival,r.date)}`);return lines.join('\n');
}
